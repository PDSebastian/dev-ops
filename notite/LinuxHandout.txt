2  sudo apt install <nume>
    # Instalează un program nou, de exemplu un joc sau un editor de text.

3  sudo apt remove <nume>
    # Șterge (dezinstalează) un program de pe calculator.

4  apt search <cuvant>
    # Caută printre programe după un cuvânt, ca la un magazin.
5  sudo adduser <nume>
    # Creează un cont nou de utilizator, ca un elev nou într-o clasă.

6  sudo groupadd <nume>
    # Creează un grup nou, ca o "echipă" în care poți pune mai mulți useri.

7  sudo usermod -aG <grup> <user>
    # Adaugă un user într-o echipă (grup), fără să-l scoată din alte echipe.

8  sudo deluser <nume>
    # Șterge un cont de utilizator de pe calculator.

9  cat /etc/passwd
    # Afișează lista cu toți userii care există pe calculator
10  ls -l
    # Arată o listă cu fișiere, și lângă fiecare, cine are voie să-l citească/scrie/ruleze.

11  chmod +x <fisier>
    # Dă unui fișier voie să fie "rulat" ca un program (nu doar citit).

12  chmod 740 <fisier>
    # Setează cu numere cine are voie ce: primul număr = tu (owner),
    # al doilea = echipa ta (grup), al treilea = restul lumii (others).

13  sudo chown <user>:<grup> <fisier>
    # Schimbă cine este "proprietarul" unui fișier.
14  comanda1 | comanda2
    # Simbolul | (pipe) trimite rezultatul primei comenzi direct la a doua,
    # ca o "conductă" prin care trece informația.

15  history | grep sudo
    # Caută în istoricul comenzilor tale doar liniile care conțin cuvântul "sudo".

16  comanda > fisier.txt
    # Salvează rezultatul unei comenzi într-un fișier, ștergând ce era înainte acolo.

17  comanda >> fisier.txt
    # La fel ca mai sus, dar ADAUGĂ la finalul fișierului, fără să șteargă nimic.

18  cat <fisier>
    # Afișează pe ecran tot ce scrie într-un fișier.

19  less <fisier>
    # Afișează un fișier mare, pagină cu pagină, ca o carte pe care o răsfoiești
