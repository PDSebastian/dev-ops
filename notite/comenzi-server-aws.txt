# Actualizează lista de pachete
sudo apt update

# Instalează Docker
sudo apt install -y docker.io

# Instalează Docker Compose v2
sudo apt install -y docker-compose-v2

# Configurează Docker să pornească automat la restart
sudo systemctl enable docker

# Pornește Docker acum
sudo systemctl start docker

# Afișează comenzile Docker disponibile
docker

# Afișează containerele care rulează
docker ps

# Adaugă utilizatorul curent în grupul docker
# După asta poți folosi Docker fără sudo
sudo usermod -aG docker $USER

# Închide sesiunea SSH pentru aplicarea grupului
exit

# Afișează istoricul comenzilor
history

# Curăță terminalul
clear

# Listează fișierele și directoarele
ls

# Mergi în directorul rădăcină
cd /

# Mergi cu un director mai sus
cd ../


# =========================
# COMENZI DOCKER UTILE
# =========================

# Vezi toate containerele, inclusiv cele oprite
docker ps -a

# Vezi imaginile Docker
docker images

# Verifică versiunea Docker
docker --version

# Verifică versiunea Docker Compose
docker compose version

# Testează dacă Docker funcționează
docker run --rm hello-world

# Pornește serviciile din compose.yaml / docker-compose.yml
docker compose up -d

# Oprește serviciile Docker Compose
docker compose down

# Vezi logurile în timp real
docker compose logs -f

# Repornește serviciile
docker compose restartasdas
